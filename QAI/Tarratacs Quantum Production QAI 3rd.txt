# STEP 1: IMPORT LIBRARIES

import itertools
import matplotlib.pyplot as plt

from qiskit import QuantumCircuit, transpile
from qiskit_aer import AerSimulator
from qiskit.visualization import plot_histogram


# STEP 2: INITIALIZE THE 3-PUZZLE

# 0 represents the blank space

initial_state = (3, 0, 1, 2)

goal_state = (0, 2, 3, 1)

# Clockwise order of positions in a 2x2 puzzle

clockwise_order = [0, 1, 3, 2]


# STEP 3: DEFINE PRODUCTION RULES

def production_rule(state, move):

    board = list(state)

    # Locate blank space
    blank = board.index(0)

    # Find blank position in clockwise order
    position = clockwise_order.index(blank)

    # Apply production rule
    if move == 0:
        next_position = clockwise_order[
            (position + 1) % 4
        ]
    else:
        next_position = clockwise_order[
            (position - 1) % 4
        ]

    # Swap blank with adjacent tile
    board[blank], board[next_position] = (
        board[next_position],
        board[blank]
    )

    return tuple(board)


# STEP 4: EXECUTE A PATH

def execute_path(path):

    state = initial_state

    for move in path:
        state = production_rule(state, int(move))

    return state


# STEP 5: GENERATE ALL POSSIBLE PATHS

depth = 3

all_paths = [
    ''.join(map(str, path))
    for path in itertools.product([0, 1], repeat=depth)
]

print("All Possible Paths:")

for path in all_paths:
    print(path, ":", execute_path(path))


# STEP 6: IDENTIFY GOAL PATH

marked_paths = [
    path for path in all_paths
    if execute_path(path) == goal_state
]

print("\nGoal Paths:", marked_paths)

if len(marked_paths) == 0:
    raise ValueError("No goal path found at depth 3.")


# STEP 7: CREATE QUANTUM ORACLE

def create_oracle(marked_paths):

    oracle = QuantumCircuit(depth)

    for path in marked_paths:

        # Convert zero bits into one controls
        for i, bit in enumerate(path):
            if bit == '0':
                oracle.x(i)

        # Apply controlled-controlled Z
        oracle.h(2)
        oracle.ccx(0, 1, 2)
        oracle.h(2)

        # Restore qubits
        for i, bit in enumerate(path):
            if bit == '0':
                oracle.x(i)

    return oracle


# STEP 8: CREATE GROVER DIFFUSER

def create_diffuser():

    diffuser = QuantumCircuit(depth)

    diffuser.h([0, 1, 2])
    diffuser.x([0, 1, 2])

    diffuser.h(2)
    diffuser.ccx(0, 1, 2)
    diffuser.h(2)

    diffuser.x([0, 1, 2])
    diffuser.h([0, 1, 2])

    return diffuser


# STEP 9: CREATE QUANTUM CIRCUIT

qc = QuantumCircuit(depth, depth)

# Initialize superposition

qc.h([0, 1, 2])

# Create oracle and diffuser

oracle = create_oracle(marked_paths)
diffuser = create_diffuser()

# Apply two Grover iterations

for i in range(2):

    qc.compose(oracle, inplace=True)
    qc.barrier()

    qc.compose(diffuser, inplace=True)
    qc.barrier()

# Measure quantum path register

qc.measure([0, 1, 2], [0, 1, 2])


# STEP 10: DISPLAY QUANTUM CIRCUIT

print("\nQuantum Circuit:")
print(qc.draw())


# STEP 11: SIMULATE QUANTUM CIRCUIT

simulator = AerSimulator()

compiled_circuit = transpile(qc, simulator)

result = simulator.run(
    compiled_circuit,
    shots=1024
).result()

counts = result.get_counts()


# STEP 12: DISPLAY RESULTS

print("\n========== QUANTUM PRODUCTION SYSTEM ==========")

print("Initial State:", initial_state)
print("Goal State:", goal_state)

print("\nMeasurement Counts:")

for key, value in counts.items():
    print(key, ":", value)


# STEP 13: FIND MOST PROBABLE PATH

best_key = max(counts, key=counts.get)

# Reverse because Qiskit displays classical bits
# in reverse order

best_path = best_key[::-1]

final_state = execute_path(best_path)

print("\n========== FINAL RESULT ==========")

print("Selected Path:", best_path)

print("Final Puzzle State:", final_state)

print("Goal State:", goal_state)

if final_state == goal_state:

    print("SUCCESS: Goal state reached!")

else:

    print("Goal state not reached.")


# STEP 14: DISPLAY FINAL PUZZLE

print("\nFinal Puzzle Configuration:")

print(final_state[0], final_state[1])
print(final_state[2], final_state[3])


# STEP 15: DISPLAY HISTOGRAM

plot_histogram(counts)

plt.title("Tarrataca's Quantum Production System")

plt.show()

print("\nExperiment Completed Successfully!")
