# -------------------- 1. IMPORT LIBRARIES --------------------


# If any library is missing, run this command in a separate notebook cell:
# %pip install pandas numpy matplotlib scikit-learn


import pandas as pd                       # Used to load and manage tabular data
import numpy as np                        # Used for numerical operations
import matplotlib.pyplot as plt           # Used to create graphs


from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
from sklearn.ensemble import RandomForestClassifier


from sklearn.metrics import (
    accuracy_score,
    precision_score,
    recall_score,
    f1_score,
    confusion_matrix,
    classification_report
)




# -------------------- 2. LOAD THE DATASET --------------------


# Read the downloaded Car Evaluation CSV file from the current folder.
# Keep car.csv in the same folder as your Jupyter Notebook.
# This dataset file does not contain a header row, so we provide
# the column names using the names parameter.
data = pd.read_csv(
    "car-evaluation-data-set.csv",
    header=None,
    names=[
        "buying",
        "maint",
        "doors",
        "persons",
        "lug_boot",
        "safety",
        "class"
    ]
)


# Display the first five rows to understand the dataset.
print("First five rows:")
print(data.head())


# Display the number of rows and columns.
print("\nDataset shape:", data.shape)


# Display all column names.
print("\nColumn names:")
print(data.columns)




# -------------------- 3. UNDERSTAND THE DATASET --------------------


print("\nDataset Information:")


# info() displays column names, data types, and non-null values.
data.info()


print("\nDataset Description:")


# describe(include='all') gives a summary of both categorical
# and numerical columns, including count, unique values, most
# frequent value, and its frequency.
print(data.describe(include="all"))


print("\nMissing Values:")


# isnull().sum() counts missing values in each column.
print(data.isnull().sum())




# -------------------- 4. HANDLE MISSING VALUES --------------------


# The Car Evaluation dataset contains categorical columns.
# Replace any missing value in a column with that column's mode
# (the most frequently occurring value).
for column in data.columns:
    if data[column].isnull().any():
        data[column] = data[column].fillna(data[column].mode()[0])


print("\nMissing Values After Handling:")
print(data.isnull().sum())




# -------------------- 5. DISPLAY CLASS DISTRIBUTION --------------------


# The class column contains the car evaluation category:
# unacc, acc, good, or vgood.
# value_counts() counts how many records belong to each category.
print("\nClass Distribution:")
print(data["class"].value_counts())




# -------------------- 6. ENCODE CATEGORICAL DATA --------------------


# Machine learning models work with numerical values.
# LabelEncoder converts text categories into integer labels.
# Example: text categories such as low, med, high, and vhigh
# are converted into numbers.
#
# We store one encoder for each column so that the mapping
# between original text and encoded numbers is retained.
label_encoders = {}


for column in data.columns:


    # Create a separate LabelEncoder for the current column.
    encoder = LabelEncoder()


    # Learn the unique text categories and convert them to numbers.
    data[column] = encoder.fit_transform(data[column])


    # Save the encoder so we can refer to it later if needed.
    label_encoders[column] = encoder


print("\nEncoded Dataset:")
print(data.head())




# -------------------- 7. SEPARATE FEATURES AND TARGET --------------------


# X contains the six input columns used to predict car acceptability.
# The class column is removed from X because it is the value to predict.
X = data.drop("class", axis=1)


# y contains the target labels from the class column.
y = data["class"]


print("\nFeatures Shape:", X.shape)
print("Target Shape:", y.shape)




# -------------------- 8. SPLIT THE DATASET --------------------


# Divide the dataset into training and testing parts.
# test_size=0.20 means 20% of the data is used for testing
# and the remaining 80% is used for training.
# random_state=42 makes the split reproducible.
# stratify=y keeps the class proportions similar in both sets.
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42,
    stratify=y
)


print("\nTraining data shape:", X_train.shape)
print("Testing data shape:", X_test.shape)




# -------------------- 9. CREATE AND TRAIN RANDOM FOREST MODEL --------------------


# Random Forest is an ensemble learning algorithm.
# It builds multiple decision trees and combines their predictions.
#
# n_estimators=100 means the model creates 100 decision trees.
# random_state=42 makes the model's random operations reproducible.
model = RandomForestClassifier(
    n_estimators=100,
    random_state=42
)


# fit() trains the Random Forest using the training features
# and their correct class labels.
model.fit(X_train, y_train)


print("\nRandom Forest model trained successfully.")




# -------------------- 10. PREDICT THE TEST DATA --------------------


# predict() uses the trained model to predict car classes
# for the unseen test data.
y_pred = model.predict(X_test)


print("\nFirst 20 predicted values:")
print(y_pred[:20])


print("\nFirst 20 actual values:")
print(y_test.iloc[:20].to_numpy())




# -------------------- 11. CALCULATE EVALUATION METRICS --------------------


# Accuracy: proportion of all predictions that are correct.
accuracy = accuracy_score(y_test, y_pred)


# Precision: among predictions for a class, how many were correct.
# average='weighted' calculates the metric for each class and
# combines them using the number of actual samples in each class.
precision = precision_score(
    y_test,
    y_pred,
    average="weighted",
    zero_division=0
)


# Recall: among actual samples of a class, how many were identified.
recall = recall_score(
    y_test,
    y_pred,
    average="weighted",
    zero_division=0
)


# F1-score combines precision and recall into one measure.
f1 = f1_score(
    y_test,
    y_pred,
    average="weighted",
    zero_division=0
)


print("\nModel Evaluation:")
print("Accuracy:", accuracy)
print("Precision:", precision)
print("Recall:", recall)
print("F1 Score:", f1)




# -------------------- 12. DISPLAY CLASSIFICATION REPORT --------------------


# The classification report shows precision, recall, F1-score,
# and support (number of actual test samples) for each class.
# target_names displays the original class names instead of numbers.
class_names = label_encoders["class"].classes_


print("\nClassification Report:")
print(
    classification_report(
        y_test,
        y_pred,
        labels=np.arange(len(class_names)),
        target_names=class_names,
        zero_division=0
    )
)




# -------------------- 13. DISPLAY CONFUSION MATRIX --------------------


# The confusion matrix compares actual class labels with predicted labels.
# Correct predictions appear along the main diagonal.
cm = confusion_matrix(y_test, y_pred)


print("\nConfusion Matrix:")
print(cm)


# Display the confusion matrix as a graph.
plt.figure(figsize=(7, 5))
plt.imshow(cm, cmap="Blues")
plt.title("Confusion Matrix - Random Forest")
plt.xlabel("Predicted Class")
plt.ylabel("Actual Class")
plt.colorbar()


# Add the numeric count inside every cell.
for i in range(cm.shape[0]):
    for j in range(cm.shape[1]):
        plt.text(j, i, cm[i, j], ha="center", va="center")


# Show the original category names on both axes.
plt.xticks(range(len(class_names)), class_names, rotation=45)
plt.yticks(range(len(class_names)), class_names)
plt.tight_layout()
plt.show()




# -------------------- 14. DISPLAY FEATURE IMPORTANCE --------------------


# Random Forest calculates how useful each input feature was
# for making predictions.
# feature_importances_ contains one importance value per feature.
importance = pd.Series(
    model.feature_importances_,
    index=X.columns
)


# Sort the values from highest importance to lowest.
importance = importance.sort_values(ascending=False)


print("\nFeature Importance:")
print(importance)




# -------------------- 15. PLOT FEATURE IMPORTANCE --------------------


# A bar chart makes it easier to compare the importance
# of the six input features.
plt.figure(figsize=(8, 5))


# Plot the feature importance values as horizontal bars.
importance.sort_values().plot(kind="barh")


plt.title("Random Forest Feature Importance")
plt.xlabel("Importance")
plt.ylabel("Features")
plt.tight_layout()
plt.show()




# -------------------- 16. FINAL RESULT --------------------


print("\nPractical completed successfully.")
print("Random Forest Accuracy:", round(accuracy * 100, 2), "%")

PRACTICAL EXPLNATION:

### 1. Aim
To implement the Random Forest Classifier, an Ensemble Learning algorithm, to predict the acceptability and safety of cars based on their features and evaluate the model's performance using classification metrics.

### 2. Description
This practical uses the Car Evaluation dataset, which contains information about cars based on attributes such as buying price, maintenance cost, number of doors, passenger capacity, luggage boot size, and safety.

The dataset is first loaded and explored to understand its structure, data types, class distribution, and missing values. Missing categorical values are handled using the mode.

Since machine learning algorithms require numerical input, categorical features are converted into numerical values using LabelEncoder.

Next, the input features and target class are separated. The dataset is divided into training and testing sets, where 80% of the data is used for training and 20% for testing.

A Random Forest Classifier is then implemented and trained. Random Forest is an ensemble learning algorithm that combines multiple decision trees to make predictions and improve classification performance.

The trained model predicts the acceptability of cars into four categories: Unacceptable, Acceptable, Good, and Very Good.

The model is evaluated using the following classification metrics:
- **Accuracy:** Measures the percentage of correctly classified records.
- **Precision:** Measures how many predicted class labels are correct.
- **Recall:** Measures how many actual class records are correctly identified.
- **F1-Score:** Combines precision and recall into a single metric.
- **Confusion Matrix:** Compares actual and predicted class labels.

Finally, a feature importance analysis is performed to identify which car attributes contribute most to the predictions. The results are visualized using a bar chart.

### 3. Libraries Used
- **Pandas:** Used for loading, cleaning, and manipulating the dataset.
- **NumPy:** Used for numerical operations and handling array data.
- **Matplotlib:** Used for visualizing the confusion matrix and feature importance.
- **Scikit-learn:** Used for data preprocessing, model training, and evaluation.
  - `train_test_split`: Divides the dataset into training and testing sets.
  - `LabelEncoder`: Converts categorical values into numerical labels.
  - `RandomForestClassifier`: Implements the Random Forest classification algorithm.
  - `accuracy_score`: Calculates classification accuracy.
  - `precision_score`: Calculates precision.
  - `recall_score`: Calculates recall.
  - `f1_score`: Calculates the F1-score.
  - `confusion_matrix`: Generates the confusion matrix.
  - `classification_report`: Displays detailed classification performance metrics.

### 4. Result
Successfully implemented the Random Forest Classifier on the Car Evaluation dataset to predict car acceptability. Evaluated the model using accuracy, precision, recall, F1-score, and confusion matrix. Also identified and visualized the importance of different car features in the classification process.
