# -------------------- 1. INSTALL / IMPORT LIBRARIES --------------------


# If any library is missing, run this command in a separate notebook cell:
# %pip install pandas numpy matplotlib scikit-learn


import numpy as np                       # Used for numerical operations
import pandas as pd                      # Used to load and handle tabular data
import matplotlib.pyplot as plt          # Used to create graphs


from tkinter import Tk                   # Used to open a file selection window
from tkinter.filedialog import askopenfilename
from sklearn.preprocessing import StandardScaler  # Standardizes feature values
from sklearn.decomposition import PCA              # Performs PCA




# -------------------- 2. UPLOAD / SELECT THE WINE DATASET --------------------


# STEP 2: LOAD THE DATASET


# Read the downloaded CSV file from the current folder.
df = pd.read_csv("wine.csv")


# Display first five rows.
print("First five rows:")
print(df.head())


# Display number of rows and columns.
print("Dataset shape:", df.shape)


# Display column names.
print("Column names:")
print(df.columns)






# -------------------- 3. UNDERSTAND AND CHECK THE DATASET --------------------


print("\nDataset Information:")
df.info()  # Shows column names, data types, and non-null values


print("\nMissing Values:")
print(df.isnull().sum())  # Counts missing values in each column


# Fill missing numeric values with the mean of their respective columns.
# This keeps the rows instead of deleting them.
numeric_columns = df.select_dtypes(include=np.number).columns
df[numeric_columns] = df[numeric_columns].fillna(df[numeric_columns].mean())


# Check missing values again after handling them
print("\nMissing Values After Handling:")
print(df.isnull().sum())




# -------------------- 4. SEPARATE FEATURES AND TARGET --------------------


# The Wine dataset has 13 input features and one target/class column.
# The last column is treated as the target label.
X = df.iloc[:, :-1]  # All columns except the last one
y = df.iloc[:, -1]   # Only the last column


print("\nFeatures Shape:", X.shape)
print("Target Shape:", y.shape)


# PCA requires numeric input features.
# Convert feature columns to numeric values if they were read as text.
X = X.apply(pd.to_numeric, errors="coerce")


# Fill any values that could not be converted, using the column mean.
X = X.fillna(X.mean())




# -------------------- 5. STANDARDIZE THE FEATURES --------------------


# StandardScaler changes the feature values so that each feature
# has approximately mean 0 and standard deviation 1.
# Standardization is important because PCA is affected by feature scales.


scaler = StandardScaler()


# fit_transform() learns the mean and standard deviation,
# then standardizes the feature values.
X_scaled = scaler.fit_transform(X)


print("\nFirst 5 rows of standardized data:")
print(X_scaled[:5])




# -------------------- 6. APPLY PRINCIPAL COMPONENT ANALYSIS --------------------


# n_components=2 means we want to reduce the 13 features to 2 components.
pca = PCA(n_components=2)


# PCA learns the principal components and transforms the data
# from 13 dimensions into 2 dimensions.
X_pca = pca.fit_transform(X_scaled)


print("\nOriginal Data Shape:", X.shape)
print("Reduced Data Shape:", X_pca.shape)




# -------------------- 7. DISPLAY EXPLAINED VARIANCE --------------------


# explained_variance_ratio_ tells us how much information (variance)
# is represented by each principal component.
print("\nExplained Variance Ratio:")
print(pca.explained_variance_ratio_)


# Add both values to find the total variance retained by two components.
print("\nTotal Variance Explained:")
print(pca.explained_variance_ratio_.sum())




# -------------------- 8. CREATE A PCA DATAFRAME --------------------


# Store the two principal components in a new DataFrame.
pca_df = pd.DataFrame(
    X_pca,
    columns=["Principal Component 1", "Principal Component 2"]
)


# Add the original class labels so they can be used in the graph.
pca_df["Target"] = y.values


print("\nPCA DataFrame:")
print(pca_df.head())




# -------------------- 9. VISUALIZE THE PCA RESULT --------------------


# Create a scatter plot of the two principal components.
plt.figure(figsize=(8, 6))


scatter = plt.scatter(
    X_pca[:, 0],       # First principal component on the x-axis
    X_pca[:, 1],       # Second principal component on the y-axis
    c=pd.factorize(y)[0],  # Give each target class a different color
    cmap="viridis"
)


plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.title("PCA - Wine Dataset")
plt.colorbar(scatter, label="Target Class")
plt.grid(True)
plt.show()




# -------------------- 10. DISPLAY FINAL RESULT --------------------


print("\nPCA completed successfully.")
print("Original number of features:", X.shape[1])
print("Number of features after PCA:", X_pca.shape[1])
print("Total variance retained:", round(pca.explained_variance_ratio_.sum() * 100, 2), "%")

To apply the Principal Component Analysis (PCA) algorithm for dimensionality reduction on the Wine dataset, reducing 13 original features to 2 principal components while retaining maximum possible information and visualizing the results.

### 2. Description
This practical uses the Wine dataset, which contains different chemical measurements of wine, such as alcohol, ash, magnesium, and other attributes.

The dataset is first loaded and explored to understand its structure, data types, and missing values. Missing numerical values are handled using the mean.
