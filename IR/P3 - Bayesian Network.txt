import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, classification_report
from sklearn.metrics import confusion_matrix, ConfusionMatrixDisplay

from pgmpy.models import DiscreteBayesianNetwork
from pgmpy.estimators import BayesianEstimator
from pgmpy.inference import VariableElimination


# 1. LOAD CSV DATASET

df = pd.read_csv("heart_disease.csv")

print("DATASET LOADED SUCCESSFULLY")

print("\nFIRST FIVE RECORDS:")
print(df.head())

print("\nDATASET SHAPE:")
print(df.shape)

print("\nCOLUMN NAMES:")
print(df.columns.tolist())


# 2. SELECT MEDICAL ATTRIBUTES

df = df[['age', 'sex', 'cp', 'thalach', 'oldpeak', 'num']].copy()

# Convert columns to numeric
df = df.apply(pd.to_numeric, errors='coerce')

# Remove missing values
df.dropna(inplace=True)

# Convert target into binary classification
# 0 = No Heart Disease
# 1 = Heart Disease

df['target'] = (df['num'] > 0).astype(int)

df.drop(columns=['num'], inplace=True)

print("\nTARGET DISTRIBUTION:")
print(df['target'].value_counts())


# 3. DISCRETIZATION

continuous_columns = ['age', 'thalach', 'oldpeak']

for column in continuous_columns:

    df[column] = pd.cut(
        df[column],
        bins=3,
        labels=False,
        include_lowest=True
    )

df = df.astype(int)

print("\nDISCRETIZED DATA:")
print(df.head())


# 4. TRAIN-TEST SPLIT

train_data, test_data = train_test_split(
    df,
    test_size=0.2,
    random_state=42,
    stratify=df['target']
)

print("\nTRAINING RECORDS:", len(train_data))
print("TESTING RECORDS:", len(test_data))


# 5. CONSTRUCT BAYESIAN NETWORK

model = DiscreteBayesianNetwork([
    ('age', 'target'),
    ('sex', 'target'),
    ('cp', 'target'),
    ('thalach', 'target'),
    ('oldpeak', 'target')
])

print("\nBAYESIAN NETWORK STRUCTURE:")
print(model.edges())


# 6. TRAIN BAYESIAN NETWORK

# Create Bayesian Estimator
estimator = BayesianEstimator(
    model,
    train_data
)

# Estimate conditional probability distributions
# Pass prior parameters to get_parameters()

cpds = estimator.get_parameters(
    prior_type='BDeu',
    equivalent_sample_size=10
)

# Add CPDs to the network
model.add_cpds(*cpds)

print("\nBAYESIAN NETWORK TRAINED SUCCESSFULLY!")

# Validate model
print("MODEL VALID:", model.check_model())


# 7. DISPLAY CONDITIONAL PROBABILITY TABLES

print("\nCONDITIONAL PROBABILITY TABLES:")

for cpd in model.get_cpds():
    print("\n", cpd)


# 8. BAYESIAN INFERENCE

inference = VariableElimination(model)

# Select one test patient
patient = test_data.iloc[0]

evidence = {
    'age': int(patient['age']),
    'sex': int(patient['sex']),
    'cp': int(patient['cp']),
    'thalach': int(patient['thalach']),
    'oldpeak': int(patient['oldpeak'])
}

print("\nPATIENT MEDICAL INFORMATION:")
print(evidence)

# Calculate probability
result = inference.query(
    variables=['target'],
    evidence=evidence,
    show_progress=False
)

print("\nDIAGNOSIS PROBABILITY:")
print(result)

probability = result.values[1]

print("\nHEART DISEASE PROBABILITY:", probability)

if probability >= 0.5:
    print("PREDICTION: HEART DISEASE")
else:
    print("PREDICTION: NO HEART DISEASE")


# 9. MODEL EVALUATION

y_actual = []
y_predicted = []

for _, patient in test_data.iterrows():

    evidence = {
        'age': int(patient['age']),
        'sex': int(patient['sex']),
        'cp': int(patient['cp']),
        'thalach': int(patient['thalach']),
        'oldpeak': int(patient['oldpeak'])
    }

    result = inference.query(
        variables=['target'],
        evidence=evidence,
        show_progress=False
    )

    prediction = int(np.argmax(result.values))

    y_actual.append(int(patient['target']))
    y_predicted.append(prediction)


# Calculate accuracy

accuracy = accuracy_score(y_actual, y_predicted)

print("\nMODEL ACCURACY:", accuracy * 100, "%")


# Classification report

print("\nCLASSIFICATION REPORT:")

print(classification_report(
    y_actual,
    y_predicted,
    zero_division=0
))


# 10. CONFUSION MATRIX

cm = confusion_matrix(y_actual, y_predicted)

display = ConfusionMatrixDisplay(
    confusion_matrix=cm,
    display_labels=['No Disease', 'Disease']
)

display.plot(cmap='Blues')

plt.title("Bayesian Network Confusion Matrix")

plt.show()


# 11. FINAL RESULT

print("\n--------------------------------")
print("PRACTICAL EXECUTED SUCCESSFULLY")
print("--------------------------------")

## Top 30 Viva Questions and Answers

### A. Basic Concepts

**Q1. What is a Bayesian Network?**

A Bayesian Network is a probabilistic graphical model that represents relationships between variables using a Directed Acyclic Graph (DAG).

It uses probability theory to represent uncertainty and make predictions.

**Q2. What is Bayes' Theorem?**

Bayes' Theorem calculates the probability of an event based on previously known information.

Formula:

P(A|B) = [P(B|A) × P(A)] / P(B)

Where:
- P(A|B): Posterior probability.
- P(B|A): Likelihood.
- P(A): Prior probability.
- P(B): Evidence.

**Q3. What is conditional probability?**

Conditional probability is the probability of an event occurring given that another event has already occurred.

Formula:

P(A|B) = P(A ∩ B) / P(B)

**Q4. What is a Directed Acyclic Graph (DAG)?**

A DAG is a graph containing directed edges without any directed cycles.

In a Bayesian Network:
- Nodes represent variables.
- Edges represent probabilistic dependencies.

**Q5. What is the difference between a Bayesian Network and a normal neural network?**

- A Bayesian Network represents probabilistic relationships between variables.
- A neural network learns patterns using interconnected neurons and weights.
- Bayesian Networks can represent uncertainty explicitly.
- Neural networks generally require numerical training and are often used for complex pattern recognition.

**Q6. Why do we use a Bayesian Network for heart disease prediction?**

Because heart disease depends on multiple medical factors. A Bayesian Network helps represent the probabilistic relationships between these factors and estimate the probability of heart disease.

**Q7. What is a prior probability?**

Prior probability is the initial probability of an event before considering new evidence.

**Q8. What is posterior probability?**

Posterior probability is the updated probability of an event after considering available evidence.

**Q25. What is Variable Elimination?**

Variable Elimination is an exact inference algorithm used to calculate probabilities in a Bayesian Network by eliminating unnecessary variables through probability factor operations.
