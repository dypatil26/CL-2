import nltk
import re
import pandas as pd
import matplotlib.pyplot as plt

from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer, WordNetLemmatizer
from collections import Counter

nltk.download('punkt')
nltk.download('punkt_tab')
nltk.download('stopwords')
nltk.download('wordnet')

document = """
Information Retrieval is an important area of computer science.
It deals with searching, retrieving and organizing information
from large collections of documents.

Text preprocessing is an essential step in Natural Language Processing.
It includes tokenization, stop word removal, stemming and lemmatization.

Machine learning techniques are widely used for information retrieval.
These techniques help users find relevant information efficiently.
"""

# Text Cleaning
cleaned_text = document.lower()
cleaned_text = re.sub(r'[^a-z\s]', '', cleaned_text)
cleaned_text = re.sub(r'\s+', ' ', cleaned_text).strip()

print("CLEANED TEXT:")
print(cleaned_text)

# Tokenization
tokens = word_tokenize(cleaned_text)
print("\nTOKENIZED WORDS:")
print(tokens)

# Stop Word Removal
stop_words = set(stopwords.words('english'))
filtered_words = [word for word in tokens if word not in stop_words]

print("\nAFTER STOP WORD REMOVAL:")
print(filtered_words)

# Stemming
stemmer = PorterStemmer()
stemmed_words = [stemmer.stem(word) for word in filtered_words]

print("\nAFTER STEMMING:")
print(stemmed_words)

# Lemmatization
lemmatizer = WordNetLemmatizer()
lemmatized_words = [lemmatizer.lemmatize(word) for word in filtered_words]

print("\nAFTER LEMMATIZATION:")
print(lemmatized_words)

# Word Frequency
word_frequency = Counter(stemmed_words)

frequency_df = pd.DataFrame(
    word_frequency.items(),
    columns=['Word', 'Frequency']
).sort_values(by='Frequency', ascending=False)

print("\nWORD FREQUENCY:")
print(frequency_df)

# Visualization
plt.figure(figsize=(10, 5))
plt.bar(frequency_df['Word'], frequency_df['Frequency'])
plt.xlabel("Words")
plt.ylabel("Frequency")
plt.title("Word Frequency After Preprocessing")
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()

### A. Basic NLP Concepts

**Q1. What is Natural Language Processing (NLP)?**

NLP stands for Natural Language Processing. It is a branch of Artificial Intelligence that enables computers to understand, process and analyze human languages.

**Q2. What is text preprocessing?**

Text preprocessing is the process of cleaning and preparing raw text before providing it to a machine learning or NLP model.

It includes:
- Text cleaning
- Tokenization
- Stopword removal
- Stemming
- Lemmatization

**Q3. Why is text preprocessing important?**

It removes unnecessary information, reduces noise and makes text easier for computers to process and analyze.

**Q4. What is text cleaning?**

Text cleaning is the process of removing unwanted characters, numbers, punctuation and extra spaces from raw text.

Tokenization is the process of splitting text into smaller units called tokens, such as words or sentences.

Example:

Stopwords are commonly occurring words that may carry limited useful information for a particular text analysis task.

Lemmatization is the process of converting words into their meaningful dictionary base form, called a lemma.

Example:

Better → Good (when the correct linguistic context and part of speech are provided).

- Stemming removes word endings using simple rules.
- Lemmatization uses linguistic knowledge to obtain meaningful base words.
- Stemming may produce words that are not dictionary words.
- Lemmatization generally produces valid dictionary words.
