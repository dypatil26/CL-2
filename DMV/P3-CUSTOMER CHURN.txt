# STEP 2: IMPORT REQUIRED LIBRARIES

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

print("Libraries imported successfully!")



# STEP 4: LOAD THE DATASET

df = pd.read_csv('Telecom_Customer_Churn.csv')

print("\nCustomer Churn Dataset:")
display(df.head())

print("Dataset loaded successfully!")


# STEP 5: EXPLORE THE DATASET

print("\nFirst Five Records:")
display(df.head())

print("\nLast Five Records:")
display(df.tail())

print("\nDataset Shape:")
print(df.shape)

print("\nColumn Names:")
print(df.columns)

print("\nData Types:")
print(df.dtypes)

print("\nStatistical Summary:")
display(df.describe(include='all'))

print("\nMissing Values:")
print(df.isnull().sum())

print("\nDuplicate Records:")
print(df.duplicated().sum())


# STEP 6: HANDLE MISSING VALUES

df.columns = df.columns.str.strip()

if 'TotalCharges' in df.columns:
    df['TotalCharges'] = pd.to_numeric(
        df['TotalCharges'], errors='coerce'
    )

numeric_columns = df.select_dtypes(
    include=np.number
).columns

categorical_columns = df.select_dtypes(
    include=['object', 'category']
).columns

for col in numeric_columns:
    df[col] = df[col].fillna(df[col].median())

for col in categorical_columns:
    if df[col].isnull().any() and not df[col].mode().empty:
        df[col] = df[col].fillna(df[col].mode()[0])

print("\nMissing Values After Handling:")
print(df.isnull().sum())


# STEP 7: REMOVE DUPLICATE RECORDS

df = df.drop_duplicates()

if 'customerID' in df.columns:
    df = df.drop_duplicates(
        subset=['customerID'],
        keep='first'
    )

print("\nDuplicate Records After Cleaning:")
print(df.duplicated().sum())

print("Dataset Shape:", df.shape)


# STEP 8: STANDARDIZE INCONSISTENT DATA

for col in df.select_dtypes(include='object').columns:
    df[col] = df[col].str.strip()

for col in ['gender', 'Partner', 'Dependents',
            'PhoneService', 'PaperlessBilling', 'Churn']:

    if col in df.columns:
        df[col] = df[col].str.title()

print("\nStandardized Dataset:")
display(df.head())


# STEP 9: CONVERT DATA TYPES

if 'TotalCharges' in df.columns:
    df['TotalCharges'] = pd.to_numeric(
        df['TotalCharges'], errors='coerce'
    )

if 'tenure' in df.columns:
    df['tenure'] = pd.to_numeric(
        df['tenure'], errors='coerce'
    )

if 'MonthlyCharges' in df.columns:
    df['MonthlyCharges'] = pd.to_numeric(
        df['MonthlyCharges'], errors='coerce'
    )

for col in df.select_dtypes(include=np.number).columns:
    df[col] = df[col].fillna(df[col].median())

print("\nUpdated Data Types:")
print(df.dtypes)


# STEP 10: IDENTIFY AND HANDLE OUTLIERS

numeric_columns = df.select_dtypes(
    include=np.number
).columns.tolist()

outlier_columns = [
    col for col in numeric_columns
    if col != 'SeniorCitizen'
]

for col in outlier_columns:

    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)

    IQR = Q3 - Q1

    lower_limit = Q1 - 1.5 * IQR
    upper_limit = Q3 + 1.5 * IQR

    outliers = (
        (df[col] < lower_limit) |
        (df[col] > upper_limit)
    ).sum()

    print(f"{col} - Outliers: {outliers}")

    df[col] = df[col].clip(
        lower_limit,
        upper_limit
    )

print("\nOutlier Handling Completed!")


# STEP 11: PERFORM FEATURE ENGINEERING

if 'tenure' in df.columns:

    df['TenureGroup'] = pd.cut(
        df['tenure'],
        bins=[-1, 12, 24, 48, 72],
        labels=['New', 'Medium', 'Long', 'Loyal']
    )

service_columns = [
    'PhoneService',
    'MultipleLines',
    'OnlineSecurity',
    'OnlineBackup',
    'DeviceProtection',
    'TechSupport',
    'StreamingTV',
    'StreamingMovies'
]

available_services = [
    col for col in service_columns
    if col in df.columns
]

if available_services:

    df['TotalServices'] = (
        df[available_services] == 'Yes'
    ).sum(axis=1)

print("\nFeature Engineered Dataset:")
display(df.head())


# STEP 12: VISUALIZE CUSTOMER CHURN

# A. Churn Distribution
if 'Churn' in df.columns:

    plt.figure(figsize=(7, 5))

    sns.countplot(data=df, x='Churn')

    plt.title('Customer Churn Distribution')
    plt.xlabel('Churn')
    plt.ylabel('Number of Customers')

    plt.show()


# B. Churn by Contract Type
if 'Contract' in df.columns and 'Churn' in df.columns:

    plt.figure(figsize=(8, 5))

    sns.countplot(
        data=df,
        x='Contract',
        hue='Churn'
    )

    plt.title('Customer Churn by Contract Type')
    plt.xlabel('Contract Type')
    plt.ylabel('Number of Customers')

    plt.xticks(rotation=30)
    plt.tight_layout()
    plt.show()


# C. Tenure Distribution
if 'tenure' in df.columns:

    plt.figure(figsize=(8, 5))

    sns.histplot(
        data=df,
        x='tenure',
        bins=20,
        kde=True
    )

    plt.title('Customer Tenure Distribution')
    plt.xlabel('Tenure (Months)')
    plt.ylabel('Number of Customers')

    plt.show()


# D. Monthly Charges vs Churn
if 'MonthlyCharges' in df.columns and 'Churn' in df.columns:

    plt.figure(figsize=(8, 5))

    sns.boxplot(
        data=df,
        x='Churn',
        y='MonthlyCharges'
    )

    plt.title('Monthly Charges vs Customer Churn')
    plt.show()


# STEP 13: CHECK CORRELATION

correlation_data = df.select_dtypes(
    include=np.number
)

plt.figure(figsize=(10, 7))

sns.heatmap(
    correlation_data.corr(),
    annot=True,
    cmap='coolwarm',
    fmt='.2f'
)

plt.title('Customer Churn Dataset Correlation Heatmap')
plt.tight_layout()
plt.show()


# STEP 14: PREPARE DATA FOR MACHINE LEARNING

if 'Churn' not in df.columns:
    raise ValueError("Churn target column not found!")

df['Churn'] = df['Churn'].map({
    'Yes': 1,
    'No': 0
})

df = df.dropna(subset=['Churn'])

df['Churn'] = df['Churn'].astype(int)

X = df.drop(
    columns=['Churn', 'customerID'],
    errors='ignore'
)

y = df['Churn']

print("\nInput Features:")
display(X.head())

print("\nTarget Variable:")
display(y.head())


# STEP 15: ENCODE CATEGORICAL VARIABLES

X = pd.get_dummies(
    X,
    drop_first=True,
    dtype=int
)

print("\nEncoded Dataset:")
display(X.head())


# STEP 16: SPLIT DATA INTO TRAINING AND TESTING SETS

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42,
    stratify=y
)

print("\nTraining Data Shape:", X_train.shape)
print("Testing Data Shape:", X_test.shape)

print("Training Target Shape:", y_train.shape)
print("Testing Target Shape:", y_test.shape)


# STEP 17: NORMALIZE / SCALE THE DATA

scale_columns = [
    col for col in
    ['tenure', 'MonthlyCharges', 'TotalCharges']
    if col in X_train.columns
]

scaler = StandardScaler()

if scale_columns:

    X_train[scale_columns] = scaler.fit_transform(
        X_train[scale_columns]
    )

    X_test[scale_columns] = scaler.transform(
        X_test[scale_columns]
    )

print("\nScaled Training Dataset:")
display(X_train.head())

print("\nScaling Completed!")


# STEP 18: EXPORT THE CLEANED DATASET

df.to_csv(
    'cleaned_customer_churn.csv',
    index=False
)

train_data = X_train.copy()
train_data['Churn'] = y_train

train_data.to_csv(
    'customer_churn_train.csv',
    index=False
)

test_data = X_test.copy()
test_data['Churn'] = y_test

test_data.to_csv(
    'customer_churn_test.csv',
    index=False
)

print("\nAll datasets exported successfully!")


# STEP 19: GENERATE FINAL REPORT

print("\n========== CUSTOMER CHURN REPORT ==========")

print("Total Customers:", len(df))

print("Total Churned Customers:",
      df['Churn'].sum())

print("Total Retained Customers:",
      (df['Churn'] == 0).sum())

print("Customer Churn Rate:",
      round(df['Churn'].mean() * 100, 2), "%")

print("\nTraining Records:", len(X_train))

print("Testing Records:", len(X_test))

print("\n===========================================")

Step 9: Identifying and Handling Outliers

Explanation:

An outlier is a data value that is significantly different from most other values in a dataset.

This step uses the Interquartile Range (IQR) method to identify and handle outliers.

1. Selecting numerical columns
numeric_columns = df.select_dtypes(    include=np.number).columns.tolist()



Selects all numerical columns and converts their names into a Python list.

2. Excluding SeniorCitizen

Excludes the SeniorCitizen column from outlier handling because it represents a binary indicator rather than a continuous measurement.

3. Calculating Q1 and Q3
Q1 = df[col].quantile(0.25)Q3 = df[col].quantile(0.75)



- Q1 is the first quartile (25th percentile).
- Q3 is the third quartile (75th percentile).
4. Calculating IQR
IQR = Q3 - Q1



IQR represents the difference between Q3 and Q1.

5. Calculating outlier limits
lower_limit = Q1 - 1.5 * IQRupper_limit = Q3 + 1.5 * IQR



Values below the lower limit or above the upper limit are considered potential outliers.

6. Counting outliers
outliers = (    (df[col] < lower_limit) |    (df[col] > upper_limit)).sum()



- < identifies values below the lower limit.
- > identifies values above the upper limit.
- | means OR.
- .sum() counts the number of outliers.
7. Handling outliers using clipping
df[col] = df[col].clip(    lower_limit,    upper_limit)



Clipping replaces values outside the allowed limits with the nearest boundary value.

For example, if the upper limit is 100 and a value is 150, it is replaced with 100.

Explanation:

Feature scaling is the process of bringing numerical features to a similar scale.

For example, tenure may range from 0 to 72, while TotalCharges may range into thousands. Different scales can affect some machine learning algorithms.

1. Selecting columns for scaling

Creates an object of the StandardScaler class.

3. Scaling training data

- fit() calculates the mean and standard deviation from the training data.
- transform() uses those values to standardize the data.

StandardScaler uses the formula:
\[
Z = \frac{X-\mu}{\sigma}
\]

Where:
- \(X\) = Original value.
- \(\mu\) = Mean of the feature.
- \(\sigma\) = Standard deviation.
- \(Z\) = Standardized value.
After scaling, features generally have a mean close to 0 and standard deviation close to 1.

4. Scaling testing data
X_test[scale_columns] = scaler.transform(    X_test[scale_columns])



Uses the same scaling parameters learned from training data.

Important: We use fit_transform() on training data and only transform() on testing data. This prevents data leakage from the testing set into the training process.
