# STEP 6: CLEAN COLUMN NAMES

def clean_column_name(column):
    column = str(column).strip().lower()
    column = re.sub(r'[^a-z0-9]+', '_', column)
    column = column.strip('_')
    return column

df.columns = [
    clean_column_name(col) for col in df.columns
]

print("\nCleaned Column Names:")
print(df.columns.tolist())


# STEP 7: IDENTIFY IMPORTANT COLUMNS

def find_column(possible_names):
    for name in possible_names:
        if name in df.columns:
            return name
    return None

price_col = find_column([
    'price', 'sale_price', 'saleprice',
    'selling_price', 'property_price'
])

area_col = find_column([
    'area', 'square_feet', 'sqft',
    'size', 'property_area', 'area_sqft',
    'living_area'
])

neighborhood_col = find_column([
    'neighborhood', 'neighbourhood',
    'location', 'locality', 'area_name'
])

property_type_col = find_column([
    'property_type', 'type', 'house_type',
    'home_type'
])

date_col = find_column([
    'sale_date', 'saledate', 'date',
    'transaction_date'
])

print("\nDetected Important Columns:")
print("Price:", price_col)
print("Area:", area_col)
print("Neighborhood:", neighborhood_col)
print("Property Type:", property_type_col)
print("Date:", date_col)


# STEP 8: HANDLE MISSING VALUES

# Convert important numeric columns into numeric format

for col in [price_col, area_col]:
    if col is not None:
        df[col] = pd.to_numeric(
            df[col].astype(str).str.replace(',', '', regex=False),
            errors='coerce'
        )

# Convert other numeric-looking columns where appropriate

for col in df.columns:
    if df[col].dtype == 'object':
        converted = pd.to_numeric(
            df[col].astype(str).str.replace(',', '', regex=False),
            errors='coerce'
        )

        if converted.notna().sum() > 0:
            if converted.notna().sum() >= 0.8 * df[col].notna().sum():
                df[col] = converted

numeric_columns = df.select_dtypes(
    include=np.number
).columns

categorical_columns = df.select_dtypes(
    include=['object', 'category']
).columns

for col in numeric_columns:
    df[col] = df[col].fillna(df[col].median())

for col in categorical_columns:
    if df[col].isnull().any() and not df[col].mode().empty:
        df[col] = df[col].fillna(df[col].mode()[0])

print("\nMissing Values After Handling:")
print(df.isnull().sum())


# STEP 9: REMOVE DUPLICATE RECORDS

df = df.drop_duplicates()

print("Updated Dataset Shape:", df.shape)


# STEP 10: STANDARDIZE CATEGORICAL DATA

for col in df.select_dtypes(include='object').columns:
    df[col] = df[col].astype(str).str.strip()

for col in [neighborhood_col, property_type_col]:
    if col is not None:
        df[col] = df[col].str.title()

print("\nStandardized Data:")
display(df.head())


# STEP 11: CONVERT DATA TYPES

    df[date_col] = pd.to_datetime(

        errors='coerce'

# STEP 13: FILTER AND SUBSET DATA

filtered_df = df.copy()

    ]

# STEP 14: FEATURE ENGINEERING

# Calculate price per square foot

if price_col is not None and area_col is not None:

    filtered_df['price_per_sqft'] = (
        filtered_df[price_col] / filtered_df[area_col]
    )

# Extract year from sale date

if date_col is not None:

    filtered_df['sale_year'] = (
        filtered_df[date_col].dt.year
    )

print("\nDataset After Feature Engineering:")
display(filtered_df.head())


# STEP 15: IDENTIFY AND HANDLE OUTLIERS

numeric_columns = filtered_df.select_dtypes(
    include=np.number
).columns.tolist()

# Exclude binary and year-like columns from IQR capping

outlier_columns = [
    col for col in numeric_columns
    if col not in ['sale_year']
    and filtered_df[col].nunique() > 10
]

for col in outlier_columns:

    Q1 = filtered_df[col].quantile(0.25)
    Q3 = filtered_df[col].quantile(0.75)

    IQR = Q3 - Q1

    lower_limit = Q1 - 1.5 * IQR
    upper_limit = Q3 + 1.5 * IQR

    outliers = (
        (filtered_df[col] < lower_limit) |
        (filtered_df[col] > upper_limit)
    ).sum()

    print(f"{col} - Outliers: {outliers}")

    filtered_df[col] = filtered_df[col].clip(
        lower_limit,
        upper_limit
    )

print("\nOutlier Handling Completed!")

Step 15: Detecting and handling outliers
An outlier is a value that is significantly different from most other values.

For example, if most property prices are between ₹40 lakh and ₹1 crore, but one property costs ₹20 crore, it may be an outlier.
A. Identify numerical columns

Selects all numerical columns.
B. Calculate quartiles
Q1 = filtered_df[col].quantile(0.25)Q3 = filtered_df[col].quantile(0.75)



- Q1 = First quartile (25th percentile).
- Q3 = Third quartile (75th percentile).
C. Calculate IQR
IQR = Q3 - Q1



IQR stands for Interquartile Range.

It measures the spread of the middle 50% of the data.
D. Calculate outlier limits
lower = Q1 - 1.5 * IQRupper = Q3 + 1.5 * IQR



These are the standard IQR-based boundaries for identifying potential outliers.
- Values below the lower limit are potential lower outliers.
- Values above the upper limit are potential upper outliers.
E. Cap outliers
filtered_df[col] = filtered_df[col].clip(    lower=lower,    upper=upper)



clip() replaces values outside the limits with the nearest boundary.

Example:

Suppose the upper limit is 100.

An outlier value of 150 becomes 100.

This is called outlier capping or winsorization.

The code excludes sale_year and numerical columns with 10 or fewer unique values because they may represent categories or discrete values rather than continuous measurements.

Purpose: Reduce the influence of extreme numerical values on statistical analysis.
