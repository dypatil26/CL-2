# STEP 2: IMPORT REQUIRED LIBRARIES

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

print("Libraries imported successfully!")


# STEP 3: CREATE SALES DATASETS IN DIFFERENT FORMATS

# CSV Dataset
csv_data = {
    'OrderID': [101, 102, 103, 104, 105],
    'OrderDate': ['2026-01-05', '2026-01-10', '2026-01-15',
                  '2026-02-02', '2026-02-10'],
    'Product': ['Laptop', 'Mobile', 'Tablet', 'Laptop', 'Headphones'],
    'Category': ['Electronics', 'electronics', 'Electronics',
                 'Electronics', 'Accessories'],
    'Region': ['Pune', 'Mumbai', 'Delhi', 'Pune', 'Mumbai'],
    'Quantity': [2, 3, 1, 2, 5],
    'UnitPrice': [50000, 20000, 30000, 50000, 2000],
    'Discount': [0.10, 0.05, 0, 0.10, 0.05],
    'CustomerID': ['C01', 'C02', 'C03', 'C04', 'C05']
}

df_csv = pd.DataFrame(csv_data)
df_csv.to_csv('sales_data.csv', index=False)


# Excel Dataset
excel_data = {
    'OrderID': [106, 107, 108, 109, 110],
    'OrderDate': ['2026-02-15', '2026-03-01', '2026-03-05',
                  '2026-03-10', '2026-03-15'],
    'Product': ['Mobile', 'Laptop', 'Headphones', 'Tablet', 'Mobile'],
    'Category': ['Electronics', 'ELECTRONICS', 'Accessories',
                 'electronics', 'Electronics'],
    'Region': ['Delhi', 'Pune', 'Mumbai', 'Delhi', 'Pune'],
    'Quantity': [2, 1, 4, np.nan, 3],
    'UnitPrice': [20000, 50000, 2000, 30000, 20000],
    'Discount': [0.05, 0.10, 0, 0.05, np.nan],
    'CustomerID': ['C06', 'C07', 'C08', 'C09', None]
}

df_excel = pd.DataFrame(excel_data)
df_excel.to_excel('sales_data.xlsx', index=False)


# JSON Dataset
json_data = {
    'OrderID': [111, 112, 113, 114, 101],
    'OrderDate': ['2026-04-01', '2026-04-05', '2026-04-10',
                  '2026-04-15', '2026-01-05'],
    'Product': ['Laptop', 'Mobile', 'Tablet', 'Headphones', 'Laptop'],
    'Category': ['electronics', 'Electronics', 'Electronics',
                 'Accessories', 'Electronics'],
    'Region': ['Mumbai', 'Delhi', 'Pune', 'Mumbai', 'Pune'],
    'Quantity': [1, 2, 3, 5, 2],
    'UnitPrice': [50000, 20000, 30000, 2000, 50000],
    'Discount': [0.10, 0.05, 0, 0.05, 0.10],
    'CustomerID': ['C10', 'C11', 'C12', 'C13', 'C01']
}

df_json = pd.DataFrame(json_data)
df_json.to_json('sales_data.json', orient='records', indent=4)

print("CSV, Excel and JSON files created successfully!")


# STEP 4: LOAD DATA FROM CSV, EXCEL AND JSON

df_csv = pd.read_csv('sales_data.csv')
df_excel = pd.read_excel('sales_data.xlsx')
df_json = pd.read_json('sales_data.json')

print("\nCSV Data:")
display(df_csv)

print("\nExcel Data:")
display(df_excel)

print("\nJSON Data:")
display(df_json)


# STEP 5: EXPLORE THE DATASETS

print("\nFirst Five Records:")
display(df_csv.head())

print("\nDataset Shapes:")
print("CSV:", df_csv.shape)
print("Excel:", df_excel.shape)
print("JSON:", df_json.shape)

print("\nColumn Names:")
print(df_csv.columns)

print("\nData Types:")
print(df_csv.dtypes)

print("\nStatistical Summary:")
display(df_csv.describe(include='all'))

print("\nMissing Values:")
print("CSV:\n", df_csv.isnull().sum())
print("Excel:\n", df_excel.isnull().sum())
print("JSON:\n", df_json.isnull().sum())

print("\nDuplicate Records:")
print("CSV:", df_csv.duplicated().sum())
print("Excel:", df_excel.duplicated().sum())
print("JSON:", df_json.duplicated().sum())


# STEP 6: COMBINE ALL DATASETS

df = pd.concat(
    [df_csv, df_excel, df_json],
    ignore_index=True
)

print("\nCombined Sales Dataset:")
display(df)

print("Total Rows:", df.shape[0])
print("Total Columns:", df.shape[1])


# STEP 7: PERFORM DATA CLEANING

# Remove duplicate orders
df = df.drop_duplicates(subset='OrderID', keep='first')

# Handle missing values
df['Quantity'] = df['Quantity'].fillna(df['Quantity'].median())
df['UnitPrice'] = df['UnitPrice'].fillna(df['UnitPrice'].median())
df['Discount'] = df['Discount'].fillna(0)
df['CustomerID'] = df['CustomerID'].fillna('Unknown')

# Standardize text values
df['Category'] = df['Category'].str.strip().str.title()
df['Product'] = df['Product'].str.strip().str.title()
df['Region'] = df['Region'].str.strip().str.title()

# Convert data types
df['OrderDate'] = pd.to_datetime(df['OrderDate'])
df['Quantity'] = df['Quantity'].astype(int)

print("\nCleaned Dataset:")
display(df)

print("\nMissing Values After Cleaning:")
print(df.isnull().sum())

print("\nDuplicate Orders After Cleaning:")
print(df.duplicated(subset='OrderID').sum())


# STEP 8: PERFORM DATA TRANSFORMATION

# Calculate total sales after discount
df['TotalSales'] = (
    df['Quantity'] *
    df['UnitPrice'] *
    (1 - df['Discount'])
)

# Extract year and month
df['Year'] = df['OrderDate'].dt.year
df['Month'] = df['OrderDate'].dt.month_name()

print("\nTransformed Sales Dataset:")
display(df)


# STEP 9: PERFORM SALES DATA ANALYSIS

# A. Total Sales
total_sales = df['TotalSales'].sum()
print("\nTotal Sales Revenue: ₹", round(total_sales, 2))

# B. Average Order Value
average_order_value = df['TotalSales'].mean()
print("Average Order Value: ₹", round(average_order_value, 2))

# C. Total Orders
total_orders = df['OrderID'].nunique()
print("Total Orders:", total_orders)

# D. Product-wise Sales
product_sales = df.groupby('Product')['TotalSales'].sum()
print("\nProduct-wise Sales:")
display(product_sales)

# E. Region-wise Sales
region_sales = df.groupby('Region')['TotalSales'].sum()
print("\nRegion-wise Sales:")
display(region_sales)

# F. Category-wise Sales
category_sales = df.groupby('Category')['TotalSales'].sum()
print("\nCategory-wise Sales:")
display(category_sales)

# G. Monthly Sales Analysis
monthly_sales = df.groupby('Month')['TotalSales'].sum()

month_order = [
    'January', 'February', 'March', 'April',
    'May', 'June', 'July', 'August',
    'September', 'October', 'November', 'December'
]

monthly_sales = monthly_sales.reindex(
    [m for m in month_order if m in monthly_sales.index]
)

print("\nMonthly Sales:")
display(monthly_sales)


# STEP 10: CREATE DATA VISUALIZATIONS

# A. Bar Chart - Product-wise Sales
plt.figure(figsize=(8, 5))
product_sales.plot(kind='bar')
plt.title('Product-wise Sales Analysis')
plt.xlabel('Product')
plt.ylabel('Total Sales (₹)')
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()


# B. Pie Chart - Category-wise Sales
plt.figure(figsize=(7, 7))
category_sales.plot(
    kind='pie',
    autopct='%1.1f%%',
    startangle=90
)
plt.title('Category-wise Sales Distribution')
plt.ylabel('')
plt.show()


# C. Line Chart - Monthly Sales Trend
plt.figure(figsize=(9, 5))
monthly_sales.plot(kind='line', marker='o')
plt.title('Monthly Sales Trend')
plt.xlabel('Month')
plt.ylabel('Total Sales (₹)')
plt.grid(True)
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()


# D. Box Plot - Sales Distribution
plt.figure(figsize=(8, 5))
sns.boxplot(x=df['TotalSales'])
plt.title('Sales Distribution and Outliers')
plt.xlabel('Total Sales (₹)')
plt.show()


# STEP 11: SAVE THE CLEANED DATASET

df.to_csv('cleaned_sales_data.csv', index=False)
df.to_excel('cleaned_sales_data.xlsx', index=False)

print("\nCleaned sales data saved successfully!")


# STEP 12: GENERATE FINAL SALES REPORT

print("\n========== SALES ANALYSIS REPORT ==========")

print("Total Orders:", df['OrderID'].nunique())

print("Total Revenue: ₹",
      round(df['TotalSales'].sum(), 2))

print("Average Order Value: ₹",
      round(df['TotalSales'].mean(), 2))

print("\nHighest Selling Product:")
print(product_sales.idxmax())

print("\nBest Performing Region:")
print(region_sales.idxmax())

print("\nHighest Revenue Category:")
print(category_sales.idxmax())

print("\n==========================================")
