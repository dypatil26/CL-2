# STEP 2: IMPORT REQUIRED LIBRARIES


import pandas as pd
import numpy as np
import matplotlib.pyplot as plt


print("Libraries imported successfully!")








# STEP 4: LOAD THE DATASET


data = pd.read_csv(“file_name”)


print("\nAir Quality Dataset:")
display(data.head())




# STEP 5: EXPLORE THE DATASET


print("\nFirst Five Records:")
display(data.head())


print("\nLast Five Records:")
display(data.tail())


print("\nDataset Shape:")
print(data.shape)


print("\nColumn Names:")
print(data.columns.tolist())


print("\nData Types:")
print(data.dtypes)


print("\nStatistical Summary:")
display(data.describe(include='all'))


print("\nMissing Values:")
print(data.isnull().sum())




# STEP 6: IDENTIFY RELEVANT COLUMNS


# Clean column names first


data.columns = (
    data.columns.str.strip()
    .str.lower()
    .str.replace(r'[^a-z0-9]+', '_', regex=True)
    .str.strip('_')
)


def find_column(possible_names):
    for col in possible_names:
        if col in data.columns:
            return col
    return None


date_col = find_column([
    'date', 'datetime', 'date_time',
    'timestamp', 'time'
])


aqi_col = find_column([
    'aqi', 'air_quality_index',
    'airqualityindex'
])


pm25_col = find_column([
    'pm2_5', 'pm25', 'pm2_5_concentration'
])


pm10_col = find_column([
    'pm10', 'pm10_concentration'
])


co_col = find_column([
    'co', 'carbon_monoxide'
])


no2_col = find_column([
    'no2', 'nitrogen_dioxide'
])


so2_col = find_column([
    'so2', 'sulphur_dioxide', 'sulfur_dioxide'
])


o3_col = find_column([
    'o3', 'ozone'
])


pollutant_columns = [
    col for col in [
        pm25_col, pm10_col, co_col,
        no2_col, so2_col, o3_col
    ]
    if col is not None
]


print("\nDetected Columns:")
print("Date:", date_col)
print("AQI:", aqi_col)
print("Pollutants:", pollutant_columns)




# STEP 7: HANDLE MISSING VALUES AND CONVERT DATA TYPES


if date_col is not None:


    data[date_col] = pd.to_datetime(
        data[date_col],
        errors='coerce'
    )


numeric_columns = [
    col for col in [aqi_col] + pollutant_columns
    if col is not None
]


for col in numeric_columns:


    data[col] = pd.to_numeric(
        data[col],
        errors='coerce'
    )


    data[col] = data[col].fillna(
        data[col].median()
    )


if date_col is not None:
    data = data.dropna(subset=[date_col])


data = data.drop_duplicates()


print("\nMissing Values After Cleaning:")
print(data.isnull().sum())


print("\nCleaned Dataset:")
display(data.head())




# STEP 8: SORT DATA BY DATE


if date_col is not None:


    data = data.sort_values(
        by=date_col
    )


    data['year'] = data[date_col].dt.year
    data['month'] = data[date_col].dt.month
    data['month_name'] = data[date_col].dt.strftime('%b')


print("\nDate-wise Sorted Dataset:")
display(data.head())




# STEP 9: VISUALIZE OVERALL AQI TREND USING LINE PLOT


if date_col is not None and aqi_col is not None:


    plt.figure(figsize=(12, 5))


    plt.plot(
        data[date_col],
        data[aqi_col],
        marker='o',
        linestyle='-',
        color='blue',
        label='AQI'
    )


    plt.title('Overall AQI Trend Over Time')
    plt.xlabel('Date')
    plt.ylabel('Air Quality Index')


    plt.legend()
    plt.grid(True)
    plt.xticks(rotation=45)
    plt.tight_layout()


    plt.show()


else:
    print("Date or AQI column not found. Line plot skipped.")




# STEP 10: PLOT INDIVIDUAL POLLUTANT LEVELS


for col in pollutant_columns:


    if date_col is not None:


        plt.figure(figsize=(10, 4))


        plt.plot(
            data[date_col],
            data[col],
            marker='o',
            label=col.upper()
        )


        plt.title(f'{col.upper()} Levels Over Time')
        plt.xlabel('Date')
        plt.ylabel('Pollutant Level')


        plt.legend()
        plt.grid(True)
        plt.xticks(rotation=45)
        plt.tight_layout()


        plt.show()




# STEP 11: COMPARE AQI VALUES USING BAR PLOT


if date_col is not None and aqi_col is not None:


    # Group AQI by date


    daily_aqi = (
        data.groupby(data[date_col].dt.date)[aqi_col]
        .mean()
        .head(15)
    )


    plt.figure(figsize=(12, 5))


    plt.bar(
        daily_aqi.index.astype(str),
        daily_aqi.values,
        color='green',
        label='Average AQI'
    )


    plt.title('Average AQI Across Dates')
    plt.xlabel('Date')
    plt.ylabel('Average AQI')


    plt.legend()
    plt.xticks(rotation=45)
    plt.grid(axis='y')
    plt.tight_layout()


    plt.show()




# STEP 12: CREATE BOX PLOT FOR AQI DISTRIBUTION


if aqi_col is not None:


    plt.figure(figsize=(7, 5))


    plt.boxplot(
        data[aqi_col].dropna(),
        patch_artist=True
    )


    plt.title('Distribution of AQI Values')
    plt.ylabel('AQI')


    plt.grid(True)
    plt.show()




# STEP 13: CREATE SCATTER PLOTS
# Explore relationship between AQI and pollutants


if aqi_col is not None:


    for col in pollutant_columns:


        plt.figure(figsize=(7, 5))


        plt.scatter(
            data[col],
            data[aqi_col],
            alpha=0.6,
            color='red',
            label=col.upper()
        )


        plt.title(f'AQI vs {col.upper()}')
        plt.xlabel(f'{col.upper()} Level')
        plt.ylabel('AQI')


        plt.legend()
        plt.grid(True)


        plt.show()




# STEP 14: CREATE MONTHLY AQI TREND


if date_col is not None and aqi_col is not None:


    monthly_aqi = (
        data.groupby(data[date_col].dt.month)[aqi_col]
        .mean()
    )


    plt.figure(figsize=(9, 5))


    plt.plot(
        monthly_aqi.index,
        monthly_aqi.values,
        marker='o',
        color='purple',
        label='Monthly Average AQI'
    )


    plt.title('Monthly Average AQI Trend')
    plt.xlabel('Month')
    plt.ylabel('Average AQI')


    plt.legend()
    plt.grid(True)
    plt.xticks(range(1, 13))


    plt.show()




# STEP 15: CREATE SUBPLOTS


available_plot_columns = [
    col for col in [aqi_col] + pollutant_columns
    if col is not None
]


if available_plot_columns:


    fig, axes = plt.subplots(
        len(available_plot_columns),
        1,
        figsize=(10, 4 * len(available_plot_columns))
    )


    if len(available_plot_columns) == 1:
        axes = [axes]


    for ax, col in zip(axes, available_plot_columns):


        ax.hist(
            data[col].dropna(),
            bins=20,
            edgecolor='black'
        )


        ax.set_title(f'Distribution of {col.upper()}')
        ax.set_xlabel(col.upper())
        ax.set_ylabel('Frequency')


    plt.tight_layout()
    plt.show()




# STEP 16: CORRELATION HEATMAP USING MATPLOTLIB


if len(numeric_columns) >= 2:


    correlation = data[numeric_columns].corr()


    plt.figure(figsize=(8, 6))


    image = plt.imshow(
        correlation,
        cmap='coolwarm',
        interpolation='nearest'
    )


    plt.colorbar(image, label='Correlation')


    plt.xticks(
        range(len(correlation.columns)),
        correlation.columns,
        rotation=45
    )


    plt.yticks(
        range(len(correlation.columns)),
        correlation.columns
    )


    plt.title('Correlation Between AQI and Pollutants')


    for i in range(len(correlation.columns)):
        for j in range(len(correlation.columns)):


            plt.text(
                j,
                i,
                round(correlation.iloc[i, j], 2),
                ha='center',
                va='center'
            )


    plt.tight_layout()
    plt.show()




# STEP 17: EXPORT CLEANED DATASET


data.to_csv(
    'cleaned_city_air_quality.csv',
    index=False
)


print("\nCleaned dataset exported successfully!")




# STEP 18: GENERATE FINAL REPORT


print("\n========== AIR QUALITY ANALYSIS REPORT ==========")


print("Total Records:", len(data))


if aqi_col is not None:


    print("Average AQI:", round(data[aqi_col].mean(), 2))


    print("Maximum AQI:", round(data[aqi_col].max(), 2))


    print("Minimum AQI:", round(data[aqi_col].min(), 2))


for col in pollutant_columns:


    print(
        f"Average {col.upper()}:",
        round(data[col].mean(), 2)
    )


print("\nPollutants Analyzed:", pollutant_columns)


print("\nCleaned file: cleaned_city_air_quality.csv")


print("\n=================================================")
print("AQI VISUALIZATION COMPLETED SUCCESSFULLY!")

### Finding columns automatically


```python
def find_column(possible_names):
    for col in possible_names:
        if col in data.columns:
            return col
    return None
```


This function searches for a column using a list of possible names.


- `for col in possible_names`: Checks each candidate name.
- `if col in data.columns`: Checks whether the column exists.
- `return col`: Returns the first matching column.
- `return None`: Returns None if no matching column is found.


The function is used to identify:


- `date_col`: Date or timestamp column.
- `aqi_col`: Air Quality Index column.

## STEP 7: Handle Missing Values and Convert Data Types


### A. Convert date into datetime format


```python
if date_col is not None:
    data[date_col] = pd.to_datetime(
        data[date_col],
        errors='coerce'
    )
```


- `pd.to_datetime()` converts date values into datetime format.
- `errors='coerce'` converts invalid date values into `NaT` (Not a Time).
- The `if` condition ensures the operation happens only if a date column exists.


### B. Identify numerical columns


```python
numeric_columns = [
    col for col in [aqi_col] + pollutant_columns
    if col is not None
]
```


Creates a list containing AQI and all available pollutant columns.


### C. Convert numerical values


```python

    data[col] = pd.to_numeric(
        data[col],
        errors='coerce'
    )
```


- `pd.to_numeric()` converts values into numeric format.
- `errors='coerce'` converts invalid values into `NaN`.


Example:


| Original value | Converted value |
|---|---:|
| 45 | 45 |
| 78.5 | 78.5 |
| Unknown | NaN |


### D. Fill missing numerical values


```python
data[col] = data[col].fillna(
    data[col].median()
)
```


Replaces missing numerical values with the median of that column.


**Median:** The middle value after arranging the values in ascending order.


Example:


`[10, 20, 30, 40, 100]`


Median = 30.


Median is useful because it is less affected by extreme values than the mean.


### E. Remove rows with invalid dates


```python
if date_col is not None:
    data = data.dropna(subset=[date_col])
```


Removes records where the date value is missing or invalid.


### F. Remove duplicate records


```python
data = data.drop_duplicates()
```


Removes duplicate rows from the dataset.


Finally, the code displays the missing-value count and cleaned data.


**Purpose:** Prepare reliable numerical and date data for visualization.
