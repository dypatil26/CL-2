# STEP 6: IDENTIFY RELEVANT COLUMNS


# Clean column names


df.columns = (
    df.columns.str.strip()
    .str.lower()
    .str.replace(r'[^a-z0-9]+', '_', regex=True)
    .str.strip('_')
)




def find_column(possible_names):
    for col in possible_names:
        if col in df.columns:
            return col
    return None




region_col = find_column([
    'region', 'sales_region',
    'geographical_region', 'area'
])


sales_col = find_column([
    'sales_amount', 'sales',
    'total_sales', 'revenue',
    'amount', 'sales_revenue'
])


category_col = find_column([
    'product_category', 'category',
    'product_type', 'product'
])


quantity_col = find_column([
    'quantity', 'quantity_sold',
    'units_sold'
])


date_col = find_column([
    'transaction_date', 'date',
    'order_date', 'sales_date'
])


print("\nDetected Columns:")
print("Region:", region_col)
print("Sales Amount:", sales_col)
print("Product Category:", category_col)
print("Quantity:", quantity_col)
print("Transaction Date:", date_col)




# STEP 7: HANDLE MISSING VALUES AND DATA TYPES


if sales_col is not None:
    df[sales_col] = pd.to_numeric(
        df[sales_col].astype(str).str.replace(',', '', regex=False),
        errors='coerce'
    )


if quantity_col is not None:
    df[quantity_col] = pd.to_numeric(
        df[quantity_col],
        errors='coerce'
    )


if date_col is not None:
    df[date_col] = pd.to_datetime(
        df[date_col],
        errors='coerce'
    )


numeric_columns = df.select_dtypes(
    include=np.number
).columns


categorical_columns = df.select_dtypes(
    include=['object', 'category']
).columns


for col in numeric_columns:
    df[col] = df[col].fillna(df[col].median())


for col in categorical_columns:
    if df[col].isnull().any() and not df[col].mode().empty:
        df[col] = df[col].fillna(df[col].mode()[0])


if date_col is not None:
    df = df.dropna(subset=[date_col])


print("\nMissing Values After Handling:")
print(df.isnull().sum())




# STEP 8: REMOVE DUPLICATE RECORDS


df = df.drop_duplicates()


print("\nDuplicate Records After Cleaning:")
print(df.duplicated().sum())


print("Updated Dataset Shape:", df.shape)




# STEP 9: STANDARDIZE CATEGORICAL DATA


for col in [region_col, category_col]:
    if col is not None:
        df[col] = (
            df[col].astype(str)
            .str.strip()
            .str.title()
        )


print("\nStandardized Dataset:")
display(df.head())




# STEP 10: IDENTIFY VALID SALES RECORDS


if region_col is None or sales_col is None:
    raise ValueError(
        "Region or sales amount column not found. "
        "Update the column names in Step 6."
    )


df = df.dropna(
    subset=[region_col, sales_col]
)


df = df[df[sales_col] >= 0]


print("\nValid Sales Records:")
display(df.head())


print("Valid Records:", len(df))




# STEP 11: GROUP SALES BY REGION


region_sales = (
    df.groupby(region_col)[sales_col]
    .sum()
    .sort_values(ascending=False)
)


print("\nTotal Sales Amount by Region:")
display(region_sales)




# STEP 12: VISUALIZE SALES BY REGION USING BAR CHART


plt.figure(figsize=(10, 5))


plt.bar(
    region_sales.index,
    region_sales.values,
    color='skyblue',
    label='Total Sales'
)


plt.title('Total Sales by Region')
plt.xlabel('Region')
plt.ylabel('Total Sales Amount')


plt.legend()
plt.xticks(rotation=45)
plt.grid(axis='y')
plt.tight_layout()


plt.show()




# STEP 13: VISUALIZE SALES DISTRIBUTION USING PIE CHART


plt.figure(figsize=(8, 8))


plt.pie(
    region_sales.values,
    labels=region_sales.index,
    autopct='%1.1f%%',
    startangle=90
)


plt.title('Sales Distribution by Region')


plt.show()




# STEP 14: IDENTIFY TOP-PERFORMING REGION


top_region = region_sales.idxmax()
top_sales = region_sales.max()


print("\nTop Performing Region:", top_region)


print("Highest Sales Amount:", top_sales)


print("\nRegion-wise Sales Ranking:")
display(region_sales.rank(
    ascending=False,
    method='dense'
).sort_values())




# STEP 15: GROUP SALES BY REGION AND PRODUCT CATEGORY


if category_col is not None:


    region_category_sales = (
        df.groupby([region_col, category_col])[sales_col]
        .sum()
        .unstack(fill_value=0)
    )


    print("\nSales by Region and Product Category:")
    display(region_category_sales)


else:
    print("\nProduct category column not found. Grouping skipped.")




# STEP 16: CREATE STACKED BAR CHART


if category_col is not None:


    region_category_sales.plot(
        kind='bar',
        stacked=True,
        figsize=(11, 6)
    )


    plt.title('Sales by Region and Product Category')
    plt.xlabel('Region')
    plt.ylabel('Total Sales Amount')


    plt.legend(
        title='Product Category',
        bbox_to_anchor=(1.05, 1),
        loc='upper left'
    )


    plt.xticks(rotation=45)
    plt.grid(axis='y')
    plt.tight_layout()


    plt.show()




# STEP 17: CREATE GROUPED BAR CHART


if category_col is not None:


    region_category_sales.plot(
        kind='bar',
        stacked=False,
        figsize=(11, 6)
    )


    plt.title('Grouped Sales Comparison by Region and Category')
    plt.xlabel('Region')
    plt.ylabel('Total Sales Amount')


    plt.legend(
        title='Product Category',
        bbox_to_anchor=(1.05, 1),
        loc='upper left'
    )


    plt.xticks(rotation=45)
    plt.grid(axis='y')
    plt.tight_layout()


    plt.show()




# STEP 18: ADDITIONAL SALES AGGREGATION


if quantity_col is not None:


    print("\nTotal Quantity Sold by Region:")


    region_quantity = (
        df.groupby(region_col)[quantity_col]
        .sum()
        .sort_values(ascending=False)
    )


    display(region_quantity)




if category_col is not None:


    print("\nTotal Sales by Product Category:")


    category_sales = (
        df.groupby(category_col)[sales_col]
        .sum()
        .sort_values(ascending=False)
    )


    display(category_sales)




# STEP 19: MONTHLY SALES ANALYSIS


if date_col is not None:


    df['month'] = df[date_col].dt.to_period('M').astype(str)


    monthly_sales = (
        df.groupby('month')[sales_col]
        .sum()
    )


    plt.figure(figsize=(11, 5))


    plt.plot(
        monthly_sales.index,
        monthly_sales.values,
        marker='o',
        color='purple',
        label='Monthly Sales'
    )


    plt.title('Monthly Sales Trend')
    plt.xlabel('Month')
    plt.ylabel('Total Sales Amount')


    plt.legend()
    plt.xticks(rotation=45)
    plt.grid(True)
    plt.tight_layout()


    plt.show()




# STEP 20: EXPORT AGGREGATED DATASETS


region_sales.to_csv(
    'sales_by_region.csv',
    header=['Total_Sales']

if category_col is not None:


    region_category_sales.to_csv(

df.to_csv(

## Important Concepts to Remember


### 1. What is Data Aggregation?
Data aggregation is the process of collecting and summarizing data to generate meaningful information.


Example: Calculating total sales for each region.


### 2. What is GroupBy?
`groupby()` is a Pandas function used to divide data into groups based on one or more columns and perform operations such as sum, mean, count, and maximum.


### 3. What is Unstack?
`unstack()` converts a grouped index level into columns, making grouped data easier to analyze and visualize.


### 4. What is the difference between Sum and Mean?
- `sum()`: Calculates the total of values.
- `mean()`: Calculates the average of values.


### 5. What is the difference between Bar Chart and Pie Chart?
- Bar chart: Compares numerical values across categories.
- Pie chart: Shows the percentage contribution of each category to the total.


### 6. What is the difference between Stacked and Grouped Bar Charts?
- Stacked: Displays categories on top of one another.
- Grouped: Displays categories side by side.


### 7. What is idxmax()?
`idxmax()` returns the index or label corresponding to the maximum value.
