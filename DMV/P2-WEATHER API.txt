# STEP 2: IMPORT REQUIRED LIBRARIES


import requests
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from getpass import getpass


print("Libraries imported successfully!")




# STEP 3: ENTER OPENWEATHERMAP API KEY


# Get API key from https://openweathermap.org/api


API_KEY = getpass("Enter your OpenWeatherMap API Key: ")


print("API key entered successfully!")




# STEP 4: DEFINE CITIES AND API URL


cities = ['Pune', 'Mumbai', 'Delhi']


BASE_URL = "https://api.openweathermap.org/data/2.5/forecast"


print("Cities selected:", cities)




# STEP 5: FETCH WEATHER DATA FROM OPENWEATHERMAP API


weather_records = []


for city in cities:


    params = {
        'q': city,
        'appid': API_KEY,
        'units': 'metric'
    }


    response = requests.get(BASE_URL, params=params)


    if response.status_code == 200:


        data = response.json()


        print("\nWeather data fetched for:", city)


        # Extract geographical information
        latitude = data['city']['coord']['lat']
        longitude = data['city']['coord']['lon']


        # Extract weather information
        for item in data['list']:


            weather_records.append({
                'City': city,
                'DateTime': item['dt_txt'],
                'Temperature': item['main']['temp'],
                'FeelsLike': item['main']['feels_like'],
                'Humidity': item['main']['humidity'],
                'Pressure': item['main']['pressure'],
                'WindSpeed': item['wind']['speed'],
                'Rainfall': item.get('rain', {}).get('3h', 0),
                'Snowfall': item.get('snow', {}).get('3h', 0),
                'Clouds': item['clouds']['all'],
                'Weather': item['weather'][0]['description'],
                'Latitude': latitude,
                'Longitude': longitude
            })


    else:
        print("Error fetching data for", city)
        print("Status Code:", response.status_code)
        print("Message:", response.text)


print("\nTotal weather records fetched:", len(weather_records))




# STEP 6: CONVERT API RESPONSE INTO PANDAS DATAFRAME


df = pd.DataFrame(weather_records)


print("\nWeather Dataset:")
display(df.head(10))


print("\nDataset Shape:")
print(df.shape)




# STEP 7: EXPLORE THE WEATHER DATASET


# Display first five records

# STEP 8: PERFORM DATA CLEANING AND PREPROCESSING


# Remove duplicate records
df = df.drop_duplicates(subset=['City', 'DateTime'])


# Convert DateTime into datetime format
df['DateTime'] = pd.to_datetime(df['DateTime'])


# Handle missing numerical values
numeric_columns = [
    'Temperature', 'FeelsLike', 'Humidity',
    'Pressure', 'WindSpeed', 'Rainfall',
    'Snowfall', 'Clouds'
]


for column in numeric_columns:
    df[column] = df[column].fillna(df[column].median())


# Handle missing weather descriptions
df['Weather'] = df['Weather'].fillna('Unknown')


# Standardize city names
df['City'] = df['City'].str.strip().str.title()


# Check cleaned dataset
print("\nCleaned Weather Dataset:")
display(df.head())


print("\nMissing Values After Cleaning:")
print(df.isnull().sum())


print("\nDuplicate Records After Cleaning:")
print(df.duplicated().sum())




# STEP 9: PERFORM DATA TRANSFORMATION


# Extract date, day and month
df['Date'] = df['DateTime'].dt.date


df['Day'] = df['DateTime'].dt.day


df['Month'] = df['DateTime'].dt.month_name()


# Create temperature category
df['Temperature_Status'] = np.where(
    df['Temperature'] >= 30,
    'Hot',
    np.where(df['Temperature'] <= 15, 'Cold', 'Moderate')
)


print("\nTransformed Weather Dataset:")
display(df.head())




# STEP 10: PERFORM WEATHER DATA ANALYSIS
